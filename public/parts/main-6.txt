function soldNightsVilla(code,monthDate=new Date(new Date().getFullYear(),new Date().getMonth(),1)){const y=monthDate.getFullYear(),m=monthDate.getMonth(),start=new Date(y,m,1),end=new Date(y,m+1,1);let sold=0;for(const b of D.bookings){if(b.villa!==code||b.status==='cancelled')continue;const bi=new Date(b.in+'T00:00:00'),bo=new Date(b.out+'T00:00:00'),a=bi>start?bi:start,z=bo<end?bo:end;if(z>a)sold+=(z-a)/86400000}return sold}
function occupancyVilla(code,monthDate=new Date(new Date().getFullYear(),new Date().getMonth(),1)){const days=new Date(monthDate.getFullYear(),monthDate.getMonth()+1,0).getDate();return days?Math.round(soldNightsVilla(code,monthDate)/days*100):0}
