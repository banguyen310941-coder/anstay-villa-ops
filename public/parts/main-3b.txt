
async function act(fn,ok){try{await fn();if(ok)toast(ok);await refresh()}catch(e){console.error(e);toast(e.message||'Có lỗi',true)}}
function bind(){
 if(current==='crm'){$('addLead')?.addEventListener('click',()=>act(async()=>{if(!$('lName').value.trim())throw new Error('Nhập tên khách');if(MODE==='demo'){D.leads.push({id:Date.now(),name:$('lName').value.trim(),phone:$('lPhone').value.trim(),source:$('lSource').value,value:+$('lValue').value||0,status:$('lStatus').value,campaign:$('lCampaign').value.trim()});return saveLocal()}const c=await insert('customers',{full_name:$('lName').value.trim(),phone:$('lPhone').value.trim()||null});await insert('leads',{customer_id:c.id,source:$('lSource').value,campaign:$('lCampaign').value.trim()||null,status:$('lStatus').value,expected_value:+$('lValue').value||0})},'Đã lưu lead'));document.querySelectorAll('[data-del-lead]').forEach(b=>b.onclick=()=>act(async()=>MODE==='demo'?(D.leads=D.leads.filter(x=>x.id!==+b.dataset.delLead),saveLocal()):remove('leads',+b.dataset.delLead),'Đã xóa lead'))}

 if(current==='booking'){
  $('calPrev')?.addEventListener('click',()=>{CAL_MONTH=new Date(CAL_MONTH.getFullYear(),CAL_MONTH.getMonth()-1,1);render()});
  $('calNext')?.addEventListener('click',()=>{CAL_MONTH=new Date(CAL_MONTH.getFullYear(),CAL_MONTH.getMonth()+1,1);render()});
  $('calToday')?.addEventListener('click',()=>{CAL_MONTH=new Date(new Date().getFullYear(),new Date().getMonth(),1);render()});
 }
 if(current==='ops'){$('addTask')?.addEventListener('click',()=>act(async()=>{if(MODE==='demo'){D.tasks.push({id:Date.now(),villa:$('tVilla').value,type:$('tType').value,employeeId:+$('tEmployee').value||null,owner:D.employees.find(e=>e.id===+$('tEmployee').value)?.name||'',due:$('tDue').value,note:$('tNote').value.trim(),done:false});return saveLocal()}await insert('housekeeping_tasks',{villa_id:vid($('tVilla').value),task_type:$('tType').value,assigned_employee_id:+$('tEmployee').value||null,due_at:$('tDue').value?new Date($('tDue').value).toISOString():null,notes:$('tNote').value.trim()||null,status:'open'})},'Đã tạo việc'));document.querySelectorAll('[data-task]').forEach(b=>b.onclick=()=>act(async()=>{const x=D.tasks.find(t=>t.id===+b.dataset.task);if(MODE==='demo'){x.done=!x.done;return saveLocal()}await update('housekeeping_tasks',x.id,{status:x.done?'open':'completed',completed_at:x.done?null:new Date().toISOString()})},'Đã cập nhật việc'))}
 if(current==='att'){
  $('adjDate').value=day();
  $('addEmployee')?.addEventListener('click',()=>act(async()=>{if(!$('eCode').value.trim()||!$('eName').value.trim())throw new Error('Nhập mã và tên');if(MODE==='demo'){D.employees.push({id:Date.now(),code:$('eCode').value.trim(),name:$('eName').value.trim(),role:$('eRole').value,phone:$('ePhone').value.trim()});return saveLocal()}await insert('employees',{employee_code:$('eCode').value.trim(),full_name:$('eName').value.trim(),role:$('eRole').value,phone:$('ePhone').value.trim()||null,active:true})},'Đã thêm nhân viên'));
