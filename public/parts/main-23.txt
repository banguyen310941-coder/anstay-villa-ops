const LIVE_MODULES=new Set(['rates','inventoryControl','reports','websales','stock']);
let LIVE_SYNC_AT=null,LIVE_SYNCING=false,LIVE_SYNC_ERROR='';

function liveModuleSummary(k=current){
  if(k==='rates'){
    const published=(D.rateCalendar||[]).filter(x=>x.published&&!x.stopSell&&x.baseRate>0).length;
    return `${(D.rateCalendar||[]).length} ngày giá · ${published} published · ${(D.channelMappings||[]).length} mapping`;
  }
  if(k==='inventoryControl'){
    const active=(D.availabilityBlocks||[]).filter(x=>x.active).length;
    return `${active} block active · ${(D.bookingPolicies||[]).length} chính sách villa · ${(D.rateSeasons||[]).length} mùa giá`;
  }
  if(k==='reports'){
    const m=typeof REPORT_MONTH==='string'?REPORT_MONTH:mon();
    const bs=typeof bookingsForMonth==='function'?bookingsForMonth(m):D.bookings.filter(x=>String(x.in||'').slice(0,7)===m);
    const ps=typeof paymentsForMonth==='function'?paymentsForMonth(m):(D.payments||[]).filter(x=>String(x.date||'').slice(0,7)===m);
    const fs=typeof financeForMonth==='function'?financeForMonth(m):D.finance.filter(x=>String(x.date||'').slice(0,7)===m);
    return `${m} · ${bs.length} booking · ${ps.length} khoản thu · ${fs.length} dòng tài chính`;
  }
  if(k==='websales'){
    const q=(D.channelOutbox||[]).filter(x=>['pending','failed','waiting_mapping'].includes(x.status)).length;
    return `${(D.websiteRequests||[]).length} yêu cầu web · ${q} outbox chưa xong · ${(D.otaPayouts||[]).length} payout`;
  }
  if(k==='stock'){
    const open=(D.purchaseOrders||[]).filter(x=>!['received','cancelled'].includes(x.status)).length;
    return `${D.items.length} SKU · ${D.moves.length} biến động · ${(D.suppliers||[]).length} NCC · ${open} PO đang mở`;
  }
  return '';
}

function liveModuleToolbar(){
  const mode='Neon LIVE';
  const when=LIVE_SYNC_AT?new Date(LIVE_SYNC_AT).toLocaleTimeString('vi-VN'):'khi đăng nhập';
  const action=current==='inventoryControl'?'<button id="liveApiSearch">Kiểm tra Website API thật</button>':current==='websales'?'<button id="liveGatewayHealth">Kiểm tra Gateway</button>':'';
  return `<div id="liveModuleToolbar" class="note" style="display:flex;justify-content:space-between;gap:12px;align-items:center;flex-wrap:wrap">
    <div><b>● Dữ liệu vận hành trực tiếp · ${esc(mode)}</b><br><span class="muted">${esc(liveModuleSummary())} · cập nhật: ${esc(when)}${LIVE_SYNC_ERROR?' · lỗi: '+esc(LIVE_SYNC_ERROR):''}</span></div>
    <div class="actions">${action}<button id="liveRefresh" class="primary" ${LIVE_SYNCING?'disabled':''}>${LIVE_SYNCING?'Đang tải...':'Làm mới dữ liệu'}</button></div>
  </div>`;
}

async function liveModuleSync(k=current,notify=true){
  if(MODE!=='auth')return;
  if(LIVE_SYNCING)return;
  LIVE_SYNCING=true;LIVE_SYNC_ERROR='';enhanceLiveModule();
  try{
    if(k==='rates'&&typeof syncRatesRemote==='function')await syncRatesRemote();
    else if(k==='inventoryControl'){
      if(typeof syncRatesRemote==='function')await syncRatesRemote();
      if(typeof syncInventoryRemote==='function')await syncInventoryRemote();
    }else if(k==='websales'&&typeof syncWebSalesRemote==='function')await syncWebSalesRemote();
    else await loadRemote();
    LIVE_SYNC_AT=Date.now();
    if(notify)toast('Đã tải dữ liệu mới nhất từ Neon');
  }catch(e){
    console.error(e);LIVE_SYNC_ERROR=e?.message||String(e);if(notify)toast(LIVE_SYNC_ERROR,true);
  }finally{
    LIVE_SYNCING=false;render();
  }
}

async function runRealWebsiteSearch(){
  const box=$('webResults'),cin=$('webCin')?.value,cout=$('webCout')?.value,guests=+$('webGuests')?.value||1;
  if(!box||!cin||!cout||cout<=cin)return toast('Chọn ngày website hợp lệ',true);
  box.innerHTML='<p class="muted">Đang gọi API Website production...</p>';
  try{
    const r=await fetch('/api/public/search?'+new URLSearchParams({check_in:cin,check_out:cout,guests:String(guests),channel:'WEBSITE',rate_plan:'BAR'}),{cache:'no-store'});
    const data=await r.json();
    if(!r.ok)throw new Error(data?.message||data?.error||('HTTP '+r.status));
    const rows=data.results||[];
    box.innerHTML=`<div class="note"><b>Kết quả API production:</b> ${data.sellable_count||0} villa có thể bán.</div><div class="scroll"><table><tr><th>Villa</th><th>Trạng thái</th><th>Giá</th><th>Lý do</th></tr>${rows.map(x=>`<tr><td><b>${esc(x.villa_name||x.villa_code||x.villa?.name||'Villa')}</b></td><td>${x.sellable?'<span class="tag ok">Có thể bán</span>':'<span class="tag warning">Chưa bán</span>'}</td><td>${x.room_amount!=null?fmt(x.room_amount):x.room_total!=null?fmt(x.room_total):'—'}</td><td>${esc(x.reason||'—')}</td></tr>`).join('')}</table></div>`;
  }catch(e){box.innerHTML=`<div class="note warn"><b>API lỗi:</b> ${esc(e?.message||String(e))}</div>`}
}

async function checkGatewayLive(){
  try{
    const r=await fetch('/api/v1/health',{cache:'no-store'}),d=await r.json();
    if(!r.ok)throw new Error(d?.error||('HTTP '+r.status));
    alert(`Gateway ${d.version} · ${d.status}\nAPI key: ${d.configuration?.api_key}\nDatabase gateway: ${d.configuration?.server_database}\nOTA adapters: ${d.configuration?.ota_adapters}`);
  }catch(e){toast(e?.message||String(e),true)}
}

const render_v17=render;
render=function(){render_v17();enhanceLiveModule()};

document.addEventListener('click',e=>{
  const refreshBtn=e.target.closest('#liveRefresh');if(refreshBtn){liveModuleSync(current,true);return}
  if(e.target.closest('#liveApiSearch')){runRealWebsiteSearch();return}
  if(e.target.closest('#liveGatewayHealth')){checkGatewayLive();return}
  const nav=e.target.closest('#nav button[data-k]');
  if(nav&&LIVE_MODULES.has(nav.dataset.k))setTimeout(()=>liveModuleSync(nav.dataset.k,false),80);
});

setInterval(()=>{if(MODE==='auth'&&LIVE_MODULES.has(current)&&!document.hidden)liveModuleSync(current,false)},60000);
render();
