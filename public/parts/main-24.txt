async function saveProEdit(){
 if(!PRO_EDITOR)return;const {type,id}=PRO_EDITOR;
 try{
  if(type==='lead'){const x=D.leads.find(v=>v.id===id),r={status:$('peStatus').value,expected_value:+$('peValue').value||0,campaign:$('peCampaign').value.trim()||null};if(MODE==='auth')await update('leads',id,r);else Object.assign(x,{status:r.status,value:r.expected_value,campaign:r.campaign||''})}
  if(type==='booking'){const x=D.bookings.find(v=>v.id===id),s=$('peStatus').value;if(s==='cancelled'&&x.status!=='cancelled'&&!confirm('Hủy booking này?'))return;const r={status:s,guests:Math.max(1,+$('peGuests').value||1),room_revenue:Math.max(0,+$('peRoom').value||0),other_revenue:Math.max(0,+$('peOther').value||0),invoice_status:$('peInvoice').value};if(MODE==='auth')await update('bookings',id,r);else Object.assign(x,{status:r.status,guests:r.guests,room:r.room_revenue,other:r.other_revenue,invoice:r.invoice_status})}
  if(type==='finance'){const x=D.finance.find(v=>v.id===id),r={paid:$('pePaid').value==='yes',due_date:$('peDue').value||null,invoice_status:$('peInvoice').value};if(MODE==='auth')await update('finance_transactions',id,r);else Object.assign(x,{paid:r.paid,due:r.due_date,invoice:r.invoice_status})}
  $('proModal').classList.remove('open');PRO_EDITOR=null;toast('Đã lưu thay đổi');if(MODE==='auth')await refresh();else{localStorage.setItem(STORE,JSON.stringify(D));render()}

}
