function bindPayouts(){
  $('createPayout')?.addEventListener('click',async()=>{const channel=$('payChannel').value,ref=$('payRef').value.trim();if(!ref)return toast('Cần nhập mã payout',true);const val=id=>Number($(id).value||0),paid=$('payPaidAt').value;try{const {error}=await neon.rpc('admin_record_ota_payout',{p_channel_code:channel,p_payout_reference:ref,p_period_start:$('payFrom').value||null,p_period_end:$('payTo').value||null,p_gross:val('payGross'),p_commission:val('payCommission'),p_fees:val('payFees'),p_tax:val('payTax'),p_net:val('payNet'),p_paid_at:paid?paid+'T12:00:00+07:00':null,p_note:null});if(error)throw new Error(dbError(error));await payoutRefresh('Đã lưu payout OTA')}catch(e){console.error(e);toast(e?.message||String(e),true)}});
  document.querySelectorAll('[data-add-payout-line]').forEach(b=>b.onclick=async()=>{const id=+b.dataset.addPayoutLine,ext=(prompt('External reservation ID của OTA:','')||'').trim();if(!ext)return;const ask=(label)=>{const x=prompt(label,'0');return x===null?null:Number(x||0)};const gross=ask('Gross booking (VND):');if(gross===null)return;const commission=ask('Commission OTA (VND):');if(commission===null)return;const fees=ask('Phí khác (VND):');if(fees===null)return;const tax=ask('Thuế/khấu trừ (VND):');if(tax===null)return;const net=ask('Net payout của booking này (VND):');if(net===null)return;if([gross,commission,fees,tax,net].some(x=>!Number.isFinite(x)||x<0))return toast('Số tiền không hợp lệ',true);try{const {error}=await neon.rpc('admin_record_ota_payout_line',{p_payout_id:id,p_external_reservation_id:ext,p_gross:gross,p_commission:commission,p_fees:fees,p_tax:tax,p_net:net,p_note:null});if(error)throw new Error(dbError(error));await payoutRefresh('Đã thêm dòng payout')}catch(e){console.error(e);toast(e?.message||String(e),true)}});
  document.querySelectorAll('[data-match-payout-line]').forEach(b=>b.onclick=async()=>{const id=+b.dataset.matchPayoutLine,code=(prompt('Nhập mã booking ANSTAY cần ghép:','')||'').trim();if(!code)return;const bk=D.bookings.find(x=>String(x.code).toUpperCase()===code.toUpperCase());if(!bk)return toast('Không tìm thấy booking trong hệ thống',true);try{const {error}=await neon.rpc('admin_match_ota_payout_line',{p_line_id:id,p_booking_id:bk.id});if(error)throw new Error(dbError(error));await payoutRefresh('Đã ghép dòng payout với booking')}catch(e){console.error(e);toast(e?.message||String(e),true)}});
  document.querySelectorAll('[data-reconcile-payout]').forEach(b=>b.onclick=async()=>{const id=+b.dataset.reconcilePayout,p=D.otaPayouts.find(x=>x.id===id),s=payoutSums(id);if(!p)return;if(s.unmatched)return toast('Còn dòng chưa ghép booking',true);if(Math.abs(s.net-p.net)>1)return toast('Tổng net các dòng chưa khớp payout',true);if(!confirm(`Đối soát ${p.channel} ${p.ref} và ghi ${fmt(p.net)} vào sổ tiền thực thu của các booking?`))return;try{const {data,error}=await neon.rpc('admin_reconcile_ota_payout',{p_payout_id:id});if(error)throw new Error(dbError(error));await loadRemote();toast(`Đã đối soát · tạo ${data?.booking_payments_created||0} dòng tiền thu`)}catch(e){console.error(e);toast(e?.message||String(e),true)}});
}
