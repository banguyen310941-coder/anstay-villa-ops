function calcVilla(code){const bs=D.bookings.filter(b=>b.villa===code&&inMon(b.in)&&b.status!=='cancelled'),fs=D.finance.filter(f=>f.villa===code&&inMon(f.date));const bookedRoom=bs.reduce((n,b)=>n+(+b.room||0),0),room=bs.reduce((n,b)=>{const c=D.commercial?.[b.id];return n+(c?.recognized==null?(+b.room||0):(+c.recognized||0))},0),ownerBasis=bs.reduce((n,b)=>{const c=D.commercial?.[b.id];return n+(c?.ownerBase==null?(+b.room||0):(+c.ownerBase||0))},0),other=bs.reduce((n,b)=>n+(+b.other||0),0),expense=fs.filter(f=>f.dir==='out').reduce((n,f)=>n+(+f.amount||0),0),v=VILLAS.find(v=>v.code===code),owner=v?.model==='share'?ownerBasis*(v.pct/100):v?.rent||0;return {room,bookedRoom,ownerBasis,other,expense,owner,profit:room+other-expense-owner}}
