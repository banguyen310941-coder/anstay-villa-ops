function v19Metrics(){
 const today=day();
 const qty=id=>(D.moves||[]).filter(m=>String(m.item)===String(id)).reduce((n,m)=>n+(['out','issue'].includes(m.type)?-(+m.qty||0):(+m.qty||0)),0);
 const low=(D.items||[]).filter(i=>qty(i.id)<(+i.min||0)).length;
 return {
  crm:(D.leads||[]).filter(x=>x.status==='new').length,
  booking:(D.bookings||[]).filter(x=>x.in===today&&x.status!=='cancelled').length,
  ops:(D.tasks||[]).filter(x=>!['completed','cancelled'].includes(x.status||(x.done?'completed':'open'))&&x.due&&new Date(x.due)<new Date()).length,
  stock:low,
  fin:(D.finance||[]).filter(x=>x.dir==='out'&&!x.paid&&x.due&&x.due<today).length,
  websales:(D.websiteRequests||[]).filter(x=>!['confirmed','rejected','cancelled'].includes(x.status)).length,
  rates:(D.rateCalendar||[]).filter(x=>!x.published&&x.baseRate>0&&x.date>=today).length
 }
}

function v19Issues(){
 const today=day(),a=[];
 const overdueTasks=(D.tasks||[]).filter(x=>!['completed','cancelled'].includes(x.status||(x.done?'completed':'open'))&&x.due&&new Date(x.due)<new Date());
 const web=(D.websiteRequests||[]).filter(x=>!['confirmed','rejected','cancelled'].includes(x.status));
 const overdueFin=(D.finance||[]).filter(x=>x.dir==='out'&&!x.paid&&x.due&&x.due<today);
 const unrec=(D.payments||[]).filter(x=>!x.reconciled);
 const ota=(D.channelOutbox||[]).filter(x=>['failed','waiting_mapping'].includes(x.status));
 const arrivals=(D.bookings||[]).filter(x=>x.in===today&&x.status!=='cancelled');
 if(arrivals.length)a.push({level:'ok',module:'booking',title:arrivals.length+' check-in hôm nay',detail:'Mở Booking để kiểm tra khách đến và tình trạng phòng.'});
 if(web.length)a.push({level:'',module:'websales',title:web.length+' yêu cầu website cần xử lý',detail:'Kiểm tra giá, cọc và xác nhận booking.'});
 if(overdueTasks.length)a.push({level:'bad',module:'ops',title:overdueTasks.length+' việc vận hành quá hạn',detail:'Housekeeping/bảo trì đã qua deadline.'});
 if(overdueFin.length)a.push({level:'bad',module:'fin',title:overdueFin.length+' khoản chi quá hạn',detail:'Cần kiểm tra thanh toán và chứng từ.'});
 if(unrec.length)a.push({level:'',module:'booking',title:unrec.length+' khoản thu chưa đối soát',detail:'Đối chiếu ngân hàng/OTA trong sổ thu booking.'});
 if(ota.length)a.push({level:'',module:'websales',title:ota.length+' sự kiện OTA chưa hoàn tất',detail:'Kiểm tra mapping hoặc lỗi đồng bộ kênh.'});
 if(!a.length)a.push({level:'ok',module:'dash',title:'Không có cảnh báo ưu tiên',detail:'Các luồng chính hiện không có việc tồn đáng chú ý.'});
 return a
}
