async function liveModuleSync(k=current,notify=true){
  if(MODE!=='auth'){LIVE_SYNC_AT=Date.now();LIVE_SYNC_ERROR='';render();return}
  if(LIVE_SYNCING)return;
  LIVE_SYNCING=true;LIVE_SYNC_ERROR='';enhanceLiveModule();
  try{
    if(k==='rates'&&typeof syncRatesRemote==='function')await syncRatesRemote();
    else if(k==='inventoryControl'){
      if(typeof syncRatesRemote==='function')await syncRatesRemote();
      if(typeof syncInventoryRemote==='function')await syncInventoryRemote();
    }else if(k==='websales'&&typeof syncWebSalesRemote==='function')await syncWebSalesRemote();
    else await loadRemote();
    LIVE_SYNC_AT=Date.now();
    if(notify)toast('Đã tải dữ liệu mới nhất từ Neon');
  }catch(e){
    console.error(e);LIVE_SYNC_ERROR=e?.message||String(e);if(notify)toast(LIVE_SYNC_ERROR,true);
  }finally{
    LIVE_SYNCING=false;render();
  }
}

async function runRealWebsiteSearch(){
  const box=$('webResults'),cin=$('webCin')?.value,cout=$('webCout')?.value,guests=+$('webGuests')?.value||1;
  if(!box||!cin||!cout||cout<=cin)return toast('Chọn ngày website hợp lệ',true);
  box.innerHTML='<p class="muted">Đang gọi API Website production...</p>';
  try{
    const r=await fetch('/api/public/search?'+new URLSearchParams({check_in:cin,check_out:cout,guests:String(guests),channel:'WEBSITE',rate_plan:'BAR'}),{cache:'no-store'});
    const data=await r.json();
    if(!r.ok)throw new Error(data?.message||data?.error||('HTTP '+r.status));
    const rows=data.results||[];
    box.innerHTML=`<div class="note"><b>Kết quả API production:</b> ${data.sellable_count||0} villa có thể bán.</div><div class="scroll"><table><tr><th>Villa</th><th>Trạng thái</th><th>Giá</th><th>Lý do</th></tr>${rows.map(x=>`<tr><td><b>${esc(x.villa_name||x.villa_code||x.villa?.name||'Villa')}</b></td><td>${x.sellable?'<span class="tag ok">Có thể bán</span>':'<span class="tag warning">Chưa bán</span>'}</td><td>${x.room_amount!=null?fmt(x.room_amount):x.room_total!=null?fmt(x.room_total):'—'}</td><td>${esc(x.reason||'—')}</td></tr>`).join('')}</table></div>`;
  }catch(e){box.innerHTML=`<div class="note warn"><b>API lỗi:</b> ${esc(e?.message||String(e))}</div>`}
}

function hset(id,s,d){const e=$('h-'+id);if(!e)return;e.querySelector('small').textContent=d;const x=e.querySelector('.hstate');x.className='hstate '+(s==='ok'?'ok':s==='bad'?'bad':'wait');x.textContent=s==='ok'?'Hoạt động':s==='bad'?'Lỗi':'Đang kiểm tra'}
function realSettings(){
 if(current!=='settings')return;const v=$('content')?.querySelector('.view');if(!v||$('systemCenter'))return;
 const old=Array.from(v.querySelectorAll('.card')).find(c=>c.querySelector('h3')?.textContent.trim()==='Trạng thái hệ thống');old?.remove();
 v.insertAdjacentHTML('afterbegin','<div class="card" id="systemCenter" style="margin-bottom:16px"><div class="row" style="justify-content:space-between;align-items:center"><div><h3>Trung tâm kiểm tra hệ thống</h3><p class="muted">Kiểm tra trực tiếp database, Booking API và Channel Gateway.</p></div><div class="actions"><button id="exportSnap">Xuất snapshot</button><button class="primary" id="systemCheck">Kiểm tra ngay</button></div></div><div class="healthgrid">'+hbox('app','App vận hành')+hbox('auth','Phiên đăng nhập')+hbox('db','Neon Database')+hbox('booking','Booking API')+hbox('gateway','Channel Gateway')+hbox('data','Dữ liệu nghiệp vụ')+hbox('rates','Giá bán')+hbox('queue','Hàng đợi OTA')+'</div></div>');
 hset('app','ok',location.host+' · giao diện đã tải');hset('data','ok',(D.bookings||[]).length+' booking · '+(D.leads||[]).length+' lead · '+(D.tasks||[]).length+' việc');hset('rates','ok',(D.rateCalendar||[]).filter(x=>x.published&&!x.stopSell&&x.baseRate>0).length+' ngày giá published');const q=(D.channelOutbox||[]).filter(x=>['pending','failed','waiting_mapping'].includes(x.status)).length;hset('queue',q?'wait':'ok',q+' sự kiện chưa hoàn tất')
}
async function systemCheck(){
 ['auth','db','booking','gateway'].forEach(x=>hset(x,'wait','Đang kiểm tra...'));hset('auth',MODE==='auth'&&SESSION?.user?.id?'ok':'bad',MODE==='auth'?(SESSION?.user?.email||'Đã có session'):'Demo local');
 if(MODE==='auth')try{const {error}=await neon.from('villas').select('id').limit(1);if(error)throw new Error(dbError(error));hset('db','ok','Neon truy vấn thành công')}catch(e){hset('db','bad',e?.message||String(e))}else hset('db','wait','Demo local');
 const t=new Date();t.setDate(t.getDate()+1);const u=new Date(t);u.setDate(t.getDate()+1);
 try{const r=await fetch('/api/public/search?'+new URLSearchParams({check_in:day(t),check_out:day(u),guests:'2',channel:'WEBSITE',rate_plan:'BAR'}),{cache:'no-store'}),d=await r.json();if(!r.ok)throw new Error(d?.message||'HTTP '+r.status);hset('booking','ok','API 200 · '+(d.sellable_count||0)+' villa bán được ngày thử')}catch(e){hset('booking','bad',e?.message||String(e))}
 try{const r=await fetch('/api/v1/health',{cache:'no-store'}),d=await r.json();if(!r.ok)throw new Error(d?.error||'HTTP '+r.status);hset('gateway','ok','Gateway '+(d.version||'')+' · '+(d.status||'ready'))}catch(e){hset('gateway','bad',e?.message||String(e))}
}
